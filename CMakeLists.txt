cmake_minimum_required(VERSION 3.16)
set(POS_ROLE "installer" CACHE STRING "installer or runtime")
if(NOT POS_ROLE MATCHES "^(installer|runtime)$")
    message(FATAL_ERROR "POS_ROLE must be installer or runtime")
endif()
set(PROJECT_VER "0.8.3-dev")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
project(pos_${POS_ROLE})

# GCC link-time optimization for POS-owned C components only. SDK/prebuilt
# hardware libraries retain their supported build settings and ABI.
option(POS_OPTIMIZE_LINK "Optimize POS components across translation units" ON)
if(POS_OPTIMIZE_LINK)
  idf_build_get_property(pos_link_options LINK_OPTIONS)
  list(REMOVE_ITEM pos_link_options "-fno-lto")
  idf_build_set_property(LINK_OPTIONS "${pos_link_options}")
  foreach(component main pos_core pos_device pos_install pos_update pos_recovery)
    idf_component_get_property(pos_lib ${component} COMPONENT_LIB)
    target_compile_options(${pos_lib} PRIVATE -Oz -flto -fno-fat-lto-objects
      -fmerge-all-constants -fno-inline-functions-called-once -fno-inline-small-functions)
  endforeach()
  idf_build_get_property(pos_executable EXECUTABLE)
  get_target_property(pos_exe_options ${pos_executable} LINK_OPTIONS)
  list(REMOVE_ITEM pos_exe_options "-fno-lto")
  set_property(TARGET ${pos_executable} PROPERTY LINK_OPTIONS "${pos_exe_options}")
  get_filename_component(pos_tool_bin "${CMAKE_C_COMPILER}" DIRECTORY)
  file(GLOB pos_lto_dirs "${pos_tool_bin}/../libexec/gcc/xtensa-esp-elf/*")
  foreach(pos_lto_dir IN LISTS pos_lto_dirs)
    if(EXISTS "${pos_lto_dir}/liblto_plugin.so")
      target_link_options(${pos_executable} PRIVATE "-B${pos_lto_dir}/")
    endif()
  endforeach()
  target_link_options(${pos_executable} PRIVATE -flto=4 -flto-partition=one -fuse-linker-plugin -fstack-usage)
endif()
